- [ ] **Step 4: Positive control — the discriminator CAN learn a separable signal.** Write `audit_b_teamdisc.py` to instantiate `TeamIntentDiscriminator(state_dim=S, num_team_codes=6)` and train it (a few hundred steps, Adam) on a SYNTHETIC dataset where the state *encodes* the label (e.g. `state = one_hot(label) padded with noise`). 

```python
import torch, numpy as np
from ha_ctse_process.team_intent import TeamIntentDiscriminator
torch.manual_seed(0); np.random.seed(0)
S, C, N = 40, 6, 3000
labels = torch.randint(0, C, (N,))
sep = torch.zeros(N, S); sep[torch.arange(N), labels] = 4.0
sep = sep + 0.5*torch.randn(N, S)            # separable-with-noise
disc = TeamIntentDiscriminator(state_dim=S, num_team_codes=C)
opt = torch.optim.Adam(disc.parameters(), lr=1e-3)
prior = torch.ones(C)/C
for _ in range(400):
    t = disc.losses(sep, labels, prior); opt.zero_grad(); t["loss"].backward(); opt.step()
print("separable acc:", float(t["acc"]))          # expect >> 1/6, near 1.0
# negative control: label independent of state
rand = 0.5*torch.randn(N, S)
disc2 = TeamIntentDiscriminator(state_dim=S, num_team_codes=C)
opt2 = torch.optim.Adam(disc2.parameters(), lr=1e-3)
for _ in range(400):
    t2 = disc2.losses(rand, labels, prior); opt2.zero_grad(); t2["loss"].backward(); opt2.step()
print("nonseparable acc:", float(t2["acc"]))       # expect ~1/6
```
