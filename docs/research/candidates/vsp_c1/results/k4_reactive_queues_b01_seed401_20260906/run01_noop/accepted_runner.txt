#!/usr/bin/env bash
echo $$ > "/home/wu/.agent-tasks/vspc1-reactive-b01-factor-run01/pid"
START_TS=$(date +%s)
echo "=== Task 'vspc1-reactive-b01-factor-run01' started at $(date -Iseconds) ===" >> "/home/wu/.agent-tasks/vspc1-reactive-b01-factor-run01/task.log"

# Keep session alive briefly for inspection, then exit
sleep 1
